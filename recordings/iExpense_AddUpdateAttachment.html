input false AttachmentUpdateMode
input desc1 AttachDescription
input att1 AttachName
input google.com AttachUrlValue
input text1 AttachText
input AttachFilePath
input text AttachType
storeTitle currentWindow
if AttachmentUpdateMode == 'false' && currentWindow.toLowerCase().indexOf('attachments') > -1
//new attachment in attachmnets list window
assertElementPresent //table[@id='AttachmentTableLayout']
clickAndWait //button[contains(@id,'AddAttachmentButton')]
endIf
input *[contains(@id,'LineAttachments')]/button AddAttachBtnSelector
storeElementPresent //*[contains(@id,'LineAttachments')]/a addAnchorExist
if addAnchorExist
storeExpression *[contains(@id,'LineAttachments')]/a AddAttachBtnSelector
endIf
if AttachmentUpdateMode == 'false' && currentWindow.toLowerCase().indexOf('details for line') > -1
//new attachment in details window, no attachments yet
clickAndWait //${AddAttachBtnSelector}
endIf
storeElementPresent css=#ReviewGenInfoHeaderAttachments > button reviewAttachmentButton
if reviewAttachmentButton
//in review page, need to click add attachment
clickAndWait css=#ReviewGenInfoHeaderAttachments > button
endIf
click id=FileName
type id=FileName ${AttachName}
type id=AkDescription ${AttachDescription}
//For new attachment select attchment type
if AttachType== 'file' && AttachmentUpdateMode == 'false'
click id=AkAttachR1
endIf
if AttachType== 'text' && AttachmentUpdateMode == 'false'
click id=AkAttachR3
endIf
if AttachType== 'file'
type id=FileInput_oafileUpload ${AttachFilePath}
elseIf AttachType== 'text'
type id=TextInput ${AttachText}
endIf
clickAndWait id=Okay
