iExpense_lineDetailsMileageSet
input DetailApply nextButtonId
input value StartDate
input StartDateFormatted
input just1 Justification
input value NumDays
input false DetailReceiptMissing
store 0 index
assertTitle *: Details for Line *
storeElementPresent id=DetailStartDate StartDateExist
if StartDateExist
type id=DetailStartDate ${StartDateFormatted}
endIf
storeElementPresent id=DetailDays NumDaysExist
if NumDaysExist
type id=DetailDays ${NumDays}
endIf
storeElementPresent id=DetailJust1 JustificationExist
if JustificationExist
type id=DetailJust1 ${Justification}
endIf
storeElementPresent id=TripDistance TripDistanceExist
if TripDistanceExist
type id=TripDistance 1
endIf
storeElementPresent id=DailyDistance DailyDistanceExist
if DailyDistanceExist
type id=DailyDistance 1
endIf
storeElementPresent id=DistancePopList DistancePopListExist
if DistancePopListExist
click id=DistancePopList
select id=DistancePopList label=Miles
endIf
storeElementPresent id=VehicleCategory VehicleCategoryExist
if VehicleCategoryExist
click id=VehicleCategory
select id=VehicleCategory label=Private
endIf
storeElementPresent id=VehicleType VehicleTypeExist
if VehicleTypeExist
click id=VehicleType
select id=VehicleType label=Electric
endIf
storeElementPresent id=FromLocation FromLocationExist
if FromLocationExist
type id=FromLocation l1
endIf
storeElementPresent id=ToLocation ToLocationExist
if ToLocationExist
type id=ToLocation t1
endIf
storeElementPresent id=LicensePlate LicensePlateExist
if LicensePlateExist
type id=LicensePlate 1
endIf
storeElementPresent id=DetailReceiptMissing DetailReceiptMissingExist
if DetailReceiptMissingExist && DetailReceiptMissing == "true"
check id=DetailReceiptMissing DetailReceiptMissing1
elseIf DetailReceiptMissingExist && !DetailReceiptMissing == "false"
uncheck id=DetailReceiptMissing DetailReceiptMissing3
endIf
//click on calculate to check for errors
clickAndWait id=DetailCalcAmt
//Wizard step 4 : Return to Cash and Other Expenses Table
storeElementPresent //table[@id="FwkErrorBeanId"]//img[contains(@src, "error")] MileageLineErrorMessageExists
if MileageLineErrorMessageExists==true
storeText //table[@id="FwkErrorBeanId"] MileageLineErrorMessage
exitTest
endIf
clickAndWait id=${nextButtonId}
//Sometimes the calculate button does not show the errors
storeElementPresent //table[@id="FwkErrorBeanId"]//img[contains(@src, "error")] MileageLineErrorMessageExists
if MileageLineErrorMessageExists==true
storeText //table[@id="FwkErrorBeanId"] MileageLineErrorMessage
exitTest
endIf
if nextButtonId.indexOf('New') > -1
assertTitle *: Details for Line *
else
assertTitle *Expense Report: *
storeText //th[9]/span CashLines_ReimbursableColumnHeader
endIf
