iExpense_downloadAttachment
storeElementPresent id=LineAttachments viewLinkExist
//Check if we are already in the attachment window
if viewLinkExist
clickAndWait id=LineAttachments
endIf
assertTitle Attachments
input 0 attachmentRowIndex
storeText xpath=//span[contains(@id, 'AkDataType:${attachmentRowIndex}')] AttachmentType
if AttachmentType.toLowerCase() === "file"
//This will download the file
clickAndWait id=N65:AkFileName:${attachmentRowIndex}
//Thsi pause is a hack untill SM-1544 is solved
pause 1000
elseIf AttachmentType.toLowerCase().indexOf("text") > -1
clickAndWait id=N65:AkFileName:${attachmentRowIndex}
storeText xpath=//table[contains(@id, 'TextInput')]//pre AttachmentText
clickAndWait id=ReturnLink
endIf
if viewLinkExist
clickAndWait css=.xg > div:nth-child(4)
endIf
