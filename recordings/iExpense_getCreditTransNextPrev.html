iExpense_getCreditTransNextPrev
input "false" ExpensesLongScroll
input true moveNext
waitForTitle *Expense Report: Credit Card Transactions
if ${moveNext} == true
store Next buttonName
else
store Previous buttonName
endIf
echo ${buttonName}
if ExpensesLongScroll == "true"
storeElementPresent //div[@id="AvailableTrxnList:ControlBar"]//td[contains(text(),'Rows')] isNextPrevExists
else
storeElementPresent //span[@id="AvailableTrxnList"]//a[contains(text(),'${buttonName}')] isNextPrevExists
storeHtmlSource getCreaditTrans_beforeCheckingNextPrev
endIf
if !${isNextPrevExists}
exitTest
endIf
if ExpensesLongScroll == "false"
clickAndWait //span[@id="AvailableTrxnList"]//a[contains(text(),'${buttonName}')]
elseIf ExpensesLongScroll == "true" && ${moveNext}
scrollDownAndWait id=scrollDiv:AvailableTrxnList
echo "next not supported"
elseIf ExpensesLongScroll == "true" && !${moveNext}
scrollUpAndWait id=scrollDiv:AvailableTrxnList
runScript javascript{triggerPaginationEvent()}
waitForPageToLoad 8000
echo "prev not supported"
endIf
storeHtmlSource getCreaditTrans_afterClickNextPrev
storeChecked //input[@name='AvailableTrxnList:selected:0'] Select
storeText id=AvailableTrxnList:TransactionDate:0 TransactionDate
storeText id=AvailableTrxnList:Age:0 AgeDays
storeText id=AvailableTrxnList:ReceiptAmount:0 TransactionAmount
storeText id=AvailableTrxnList:OieMerchantName:0 MerchantName
storeText id=AvailableTrxnList:OieLocation:0 Location
if ExpensesLongScroll == "true"
storeText //div[@id="AvailableTrxnList:ControlBar"]//td[contains(text(),'Rows')] currentListRange
storeElementPresent //span[@id="AvailableTrxnList"]//a[contains(text(),'Next')] nextButtonExist
//There is a hidden button
storeElementPresent //span[@id="AvailableTrxnList"]//a[contains(text(),'Previous')] prevButtonExist
else
storeSelectedLabel //span[@id="AvailableTrxnList"]//select[@title="Select record set"] currentListRange
storeElementPresent //span[@id="AvailableTrxnList"]//a[contains(text(),'Next')] nextButtonExist
storeElementPresent //span[@id="AvailableTrxnList"]//a[contains(text(),'Previous')] prevButtonExist
endIf
if ${prevButtonExist} && ExpensesLongScroll == "false"
storeText //a[contains(text(),'Previous')] prevButtonLabel
endIf
if ${nextButtonExist} && ExpensesLongScroll == "false"
storeText //a[contains(text(),'Next')] nextButtonLabel
endIf
