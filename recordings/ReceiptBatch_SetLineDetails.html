ReceiptBatch_SetLineDetails
input All Other Locations LocationName
input false DetailReceiptMissing
input CompanyAttendees
input ReceiptCurrencyChoice
input DetailReceiptAmount
input StartDateFormatted
input Justification
input ExchRate
input DetailMerchantName
input AirTravelType
input AirTicketClass
input AirFromLocation
input AirToLocation
input NumberOfDays
input AccEndDateFormatted
input MealNumberAttendees
input MealAttendees
assertTitle *: Details for Line *
//The Details page/session must already be open; this partial service does not get or login.
//no need to set expense type as we assume we are displaying the correct expense type
//handle common fields
echo @@@ handle common fields
storeElementPresent //input[@id='StartDate'] | //input[@id='IPL_StartDate'] StartDateExist
if StartDateExist
type //input[@id='StartDate'] | //input[@id='IPL_StartDate'] ${StartDateFormatted}
endIf
storeElementPresent //*[@id = 'DetailReceiptAmount'] | //*[@id = 'IPL_OriginalReceiptAmount'] DetailReceiptAmountExist
storeElementPresent //span[@id = 'DetailReceiptAmount'] | //span[@id = 'IPL_OriginalReceiptAmount'] DetailReceiptAmountTextExist
if DetailReceiptAmountExist && !DetailReceiptAmountTextExist
type //input[@id = 'DetailReceiptAmount'] | //input[@id = 'IPL_OriginalReceiptAmount'] ${DetailReceiptAmount}
endIf
storeElementPresent //textarea[@id = 'Justification'] | //textarea[@id = 'IPL_Justification'] JustificationExist
if JustificationExist
type //textarea[@id = 'Justification'] | //textarea[@id = 'IPL_Justification'] ${Justification}
endIf
storeElementPresent //input[@id= 'RBLocation'] | //input[@id= 'IPL_RBLocation'] LocationNameExist
if LocationNameExist
type //input[@id= 'RBLocation'] | //input[@id= 'IPL_RBLocation'] ${LocationName}
endIf
storeElementPresent //input[@id = 'DetailMerchantName'] | //input[@id = 'IPL_MerchantName'] DetailMerchantNameExist
if DetailMerchantNameExist
storeText //span[@id = 'DetailMerchantName'] | //span[@id = 'IPL_MerchantName'] DetailMerchantNameText
endIf
if DetailMerchantNameExist && DetailMerchantNameText ==""
type //input[@id = 'DetailMerchantName'] | //input[@id = 'IPL_MerchantName'] ${DetailMerchantName}
endIf
storeElementPresent //input[@id = "ReceiptMissing"] | //input[@id = "IPL_ReceiptMissing"] | //input[@id = "DetailReceiptMissing"] DetailReceiptMissingExist
if DetailReceiptMissingExist && DetailReceiptMissing == "true"
check //input[@id = "ReceiptMissing"] | //input[@id = "IPL_ReceiptMissing"] | //input[@id = "DetailReceiptMissing"]
elseIf DetailReceiptMissingExist && DetailReceiptMissing == "false"
uncheck //input[@id = "ReceiptMissing"] | //input[@id = "IPL_ReceiptMissing"] | //input[@id = "DetailReceiptMissing"]
endIf
//Currency fields
echo @@@ Currency fields
storeElementPresent //select[@id = "ReceiptCurrencyChoice"] | //select[@id = "IPL_ReceiptCurrency"] ReceiptCurrencyChoiceExist
if ReceiptCurrencyChoiceExist
selectAndWait //select[@id = "ReceiptCurrencyChoice"] | //select[@id = "IPL_ReceiptCurrency"] ${ReceiptCurrencyChoice}
endIf
storeElementPresent //tr[@id='ReceiptCurrencyChoice__xc_']/td[3]/button hasCurrencyButton
if ReceiptCurrencyChoiceExist && hasCurrencyButton==true
clickAndWait //tr[@id='ReceiptCurrencyChoice__xc_']/td[3]/button /* Go button of currency */
endIf
storeElementPresent //input[@id= "ExchRate"] | //input[@id= "IPL_ExchangeRate"] ExchRateExist
if ExchRateExist
storeText //span[@id= "ExchRate"] | //span[@id = "IPL_ExchangeRate"] ExchRateText
endIf
if ExchRateExist && ExchRateText ==""
type //input[@id= "ExchRate"] | //input[@id= "IPL_ExchangeRate"] ${ExchRate}
endIf
//handle attachment
echo @@@ handle attachment
storeElementPresent id=LineAttachments_viewButton AttachmentExists
if !AttachmentExists
storeElementPresent //a[contains(@id, "Attachments") and contains(text(),'View')] AttachmentExists
endIf
storeElementPresent //*[@id='LineAttachments_addButton'] attachmentAddButton1
input attachmentPath
storeElementPresent //*[@id='LineAttachments']/button attachmentAddButton2
if !AttachmentExists && attachmentPath!="" && attachmentAddButton1
clickAndWait //*[@id='LineAttachments_addButton'] /* Add button */
storeHtmlSource iExpense_AfterPressAddAttachmentButton
endIf
if !AttachmentExists && attachmentPath!="" && attachmentAddButton2
clickAndWait //*[@id='LineAttachments']/button /* Add button */
endIf
storeElementPresent css=img[alt="Close Popup"] AttachmentsPopup
if !AttachmentExists && attachmentPath!="" && AttachmentsPopup
select id=attType label=File
select id=categoryList label=Receipts
type id=FileInput ${attachmentPath}
click //input[@value='Apply']
assertElementNotPresent id=dialogOkButton
storeElementPresent //input[@value='OK'] AttachmentConfirmationExists
if AttachmentConfirmationExists
click //input[@value='OK']
endIf
elseIf !AttachmentExists && attachmentPath!=""
type id=FileInput_oafileUpload ${attachmentPath}
clickAndWait id=Okay
endIf
//Airfare
echo @@@ Wizard step 4 : Expanse Details (Airfare)
storeElementPresent //select[@id="AirTravelType"] AirTravelTypeExist
if AirTravelTypeExist
select //select[@id="AirTravelType"] label=${AirTravelType}
endIf
storeElementPresent //select[@id="AirTicketClass"] AirTicketClassExist
if AirTicketClassExist
select //select[@id="AirTicketClass"] label=${AirTicketClass}
endIf
storeElementPresent //input[@id="AirFromLocation"] AirFromLocationExist
if AirFromLocationExist
type //input[@id="AirFromLocation"] ${AirFromLocation}
endIf
storeElementPresent //input[@id="AirToLocation"] AirToLocationExist
if AirToLocationExist
type //input[@id="AirToLocation"] ${AirToLocation}
endIf
//Car Rental
echo @@@ Wizard step 4 : Expanse Details (Car Rental)
//Not inputing DailyRate as it is automatically calculated from the receipt amound and number of days
storeElementPresent //input[@id = "NumberOfDays"] NumberOfDaysExist
if NumberOfDaysExist
type //input[@id = "NumberOfDays"] ${NumberOfDays}
endIf
storeElementPresent //input[@id = 'LocationName'] | //input[@id = 'IPL_LocationName'] LocationNameExist
if LocationNameExist
echo ${LocationName}
type //input[@id = 'LocationName'] | //input[@id = 'IPL_LocationName'] ${LocationName}
//Hack - move focus from location field and wait to resolve. Otherwise not working in ebs 12.2.7
click //textarea[@id = 'Justification'] | //textarea[@id = 'IPL_Justification']
pause 1000
endIf
//Entertainment
echo @@@ Wizard step 4 : Expanse Details (Entertainment)
storeElementPresent id=DFF_100102 CompanyAttendeesExist
if CompanyAttendeesExist
type id=DFF_100102 ${CompanyAttendees}
endIf
//Hotel
echo @@@ Wizard step 4 : Expanse Details (Hotel)
storeElementPresent //input[@id ="AccEndDate"] AccEndDateExist
if AccEndDateExist
type //input[@id ="AccEndDate"] ${AccEndDateFormatted}
endIf
//Miscellaneous
//Taxis
//Telephone Calls
//Meals - Must be last because of a bug with meal no attendies in EBS
storeElementPresent //input[@id="MealNumberAttendees"] MealNumberAttendeesExist
if MealNumberAttendeesExist
type //input[@id="MealNumberAttendees"] ${MealNumberAttendees}
endIf
storeElementPresent //input[@id="MealAttendees"] MealAttendeesExist
if MealAttendeesExist
//MealAttendees remains a scalar string; the native RPA employee array is not mapped here.
type //input[@id="MealAttendees"] ${MealAttendees}
endIf
//Check for policy warnings
echo @@@ Check for policy warnings
storeElementPresent //tr[contains(@id, "DetailReceiptAmount")]//button[contains(text(), "Go")] ReceiptAmountGoButtonExists
if ReceiptAmountGoButtonExists
clickAndWait //tr[contains(@id, "DetailReceiptAmount")]//button[contains(text(), "Go")]
endIf
//Hack - I repeat this section because of a bug in our ebs 12.1 which erases this specific field each time the page refreshes
storeElementPresent //input[@id="MealNumberAttendees"] MealNumberAttendeesExist
if MealNumberAttendeesExist
type //input[@id="MealNumberAttendees"] ${MealNumberAttendees}
endIf
//End of hack
storeElementPresent //table[@id="FwkErrorBeanId"]//img[contains(@src, "warn")] CashLineWarningMessageExists
store javascript{""} CashLineWarningMessage
if CashLineWarningMessageExists
storeText //table[@id="FwkErrorBeanId"] CashLineWarningMessage
endIf
//Fresh DOM flags for the RPA caller; no warning exit here because this partial service stops on Details.
storeElementPresent //table[@id="FwkErrorBeanId"]//img[contains(@src, "error")] CashLineErrorMessageExists
storeElementPresent id=orapopupcontent errorPopupExists
store javascript{""} CashLineErrorMessage
if CashLineErrorMessageExists
storeText //table[@id="FwkErrorBeanId"] CashLineErrorMessage
endIf
storeTitle currentPageTitle
//SetLineDetails stops on the Details page; no navigation is performed by this partial service.
