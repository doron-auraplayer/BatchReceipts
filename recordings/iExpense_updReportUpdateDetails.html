endIf
if CurrencyListExist  && CurrentCurrency !== ReimbCurrency&& hasCurrencyButton==true
clickAndWait //tr[@id='ReimbCurrency__xc_']/td[3]/button /* Go button of currency */
endIf
storeTitle currentTitle
//In update mode there might be confirmation screen
if currentTitle == 'Warning'
clickAndWait //button[contains(text(),'es')]
assertTitle *Expense Report: General Information*
endIf
//Type page fields value
storeElementPresent id=KffSEGMENT2 DeparmentFieldExist
if DeparmentFieldExist

endIf
type id=Purpose Conference2
pause 200

type id=OverrideApprover ${OverridingApprover}
endIf

type id=OverrideApprover ${OverridingApprover}
endIf

storeElementPresent //table[@id='FwkErrorBeanId']/tbody//img[contains(@src, 'error')] isErrorMessage

storeText //table[@id='FwkErrorBeanId'] reportErrorMessage
endIf

endIf

endIf
