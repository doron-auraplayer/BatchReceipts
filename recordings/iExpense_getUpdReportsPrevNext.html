input "false" ExpensesLongScroll
input true moveNext
input N14 ExpensesTableUpdateReportPrefix
input ReportTotalCurrency ExpensesTableReportTotalColumnName
waitForTitle Expenses Home
if ${moveNext} == true
store Next updButtonName
else
store Previous updButtonName
endIf
storeElementPresent //table[@id="ActiveReportsRN"]//a[contains(text(),'${updButtonName}')] isNextPrevExists
if !${isNextPrevExists}
exitTest
endIf
if ExpensesLongScroll == "false"
clickAndWait //table[@id="ActiveReportsRN"]//a[contains(text(),'${updButtonName}')]
elseif ExpensesLongScroll == "true" && ${moveNext}
scrollDownAndWait id=scrollDiv:ActiveReportsRN.ActiveReportsTbl
echo "next not supported"
elseif ExpensesLongScroll == "true" && !${moveNext}
scrollUpAndWait id=scrollDiv:ActiveReportsRN.ActiveReportsTbl
runScript javascript{triggerPaginationEvent()}
waitForPageToLoad 8000
echo "prev not supported"
endIf
assertTitle Expenses Home
storeText //a[contains(@id, ':ReportNumber:0')] Update Report Number
storeText //span[contains(@id, ':Date:0')] Update Report Date
storeText //span[contains(@id, ':Status:0')] Update Report Status
storeElementPresent //span[contains(@id, ':ReportTotalCurrency:0')] hasTotalCurrency
if hasTotalCurrency
storeText //span[@id='ActiveReportsRN.ActiveReportsTbl']//span[contains(@id, ':ReportTotalCurrency:0')] Update Report Total
else
storeText //span[@id='ActiveReportsRN.ActiveReportsTbl']//span[contains(@id, ':ReportTotal:0')] Update Report Total
endIf
storeText //span[contains(@id, ':Purpose:0')] Update Report Purpose
if ExpensesLongScroll == "true"
storeText //div[@id='ActiveReportsRN.ActiveReportsTbl:ControlBar']/table/tbody/tr/td[15] currentListRange
storeElementPresent //div[@id='ActiveReportsRN.ActiveReportsTbl:ControlBar']/table/tbody/tr/td[15] nextButtonExist
storeElementPresent //table[@id="ActiveReportsRN"]//a[contains(text(),'Previous')] prevButtonExist
else
storeElementPresent //table[@id="ActiveReportsRN"]//a[contains(text(),'Previous')] prevButtonExist
storeSelectedLabel //table[@id="ActiveReportsRN"]//select[@title="Select record set"] currentListRange
storeElementPresent //table[@id="ActiveReportsRN"]//a[contains(text(),'Next')] nextButtonExist
endIf
if ${prevButtonExist}  && ExpensesLongScroll == "false"
storeText //table[@id="ActiveReportsRN"]//a[contains(text(),'Previous')] prevButtonLabel
endIf
if ${nextButtonExist}  && ExpensesLongScroll == "false"
storeText //table[@id="ActiveReportsRN"]//a[contains(text(),'Next')] nextButtonLabel
endIf
