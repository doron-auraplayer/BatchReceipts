iExpense_lineDetailsPerDiemSet
input false onlyCalculate
input DetailApply nextButtonId
input Per Diem expenseType
input value StartDate
input StartDateFormatted
input just1 Justification
input LocationName
store 0 index
storeHtmlSource iExpenseLineSetStart_${index}
assertTitle *: Details for Line *
storeElementPresent id=DetailExpType expenseTypeExist
if expenseTypeExist
select id=DetailExpType label=${expenseType}
endIf
storeElementPresent id=DetailStartDate StartDateExist
if StartDateExist
type id=DetailStartDate ${StartDateFormatted}
endIf
storeElementPresent id=DetailJust JustificationExist
if JustificationExist
type id=DetailJust ${Justification}
endIf
storeElementPresent id=NumDays NumberOfDaysExist
if NumberOfDaysExist
type id=NumDays 1
endIf
storeElementPresent id=LocationName LocationNameExist
if LocationNameExist
typeAndWait id=LocationName ${LocationName}
endIf
//click on calculate and check for errors
storeHtmlSource iExpenseLineSetAfterCalculate_${index}
clickAndWait id=GeneratePerDiem
storeElementPresent //table[@id="FwkErrorBeanId"]//img[contains(@src, "error")] CashLineErrorMessageExists
storeText //table[@id="FwkErrorBeanId"] CashLineErrorMessage
if onlyCalculate == "true" || CashLineErrorMessageExists
exitTest
endIf
//Wizard step 4 : Return to Cash and Other Expenses Table
storeHtmlSource iExpenseLineSetBeforeReturn_${index}
clickAndWait id=${nextButtonId}
if nextButtonId.indexOf('New') > -1
storeHtmlSource iExpenseLineSetAfterNew_${index}
assertTitle *: Details for Line *
else
storeHtmlSource iExpenseLineSetAfterReturn_${index}
assertTitle *Expense Report: *
storeText //th[9]/span CashLines_ReimbursableColumnHeader
endIf
